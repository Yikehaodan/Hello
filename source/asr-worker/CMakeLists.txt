cmake_minimum_required(VERSION 3.16)
project(lightvoice_asr_worker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT DEFINED WHISPER_SOURCE_DIR)
  message(FATAL_ERROR "WHISPER_SOURCE_DIR must point to whisper.cpp v1.9.5")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES ON CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_SOURCE_DIR}" whisper)
if(NOT DEFINED LIGHTVOICE_WHISPER_SOURCE OR NOT EXISTS "${LIGHTVOICE_WHISPER_SOURCE}")
  message(FATAL_ERROR "Use source/build-worker.sh to generate the reviewed private Whisper source")
endif()
get_target_property(lightvoice_whisper_sources whisper SOURCES)
list(REMOVE_ITEM lightvoice_whisper_sources whisper.cpp)
list(APPEND lightvoice_whisper_sources "${LIGHTVOICE_WHISPER_SOURCE}")
set_property(TARGET whisper PROPERTY SOURCES "${lightvoice_whisper_sources}")
add_executable(lightvoice-asr-worker worker.cpp)
target_include_directories(lightvoice-asr-worker PRIVATE "${WHISPER_SOURCE_DIR}/examples")
target_link_libraries(lightvoice-asr-worker PRIVATE common whisper)
